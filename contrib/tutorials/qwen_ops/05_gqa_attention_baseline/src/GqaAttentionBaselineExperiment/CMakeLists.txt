cmake_minimum_required(VERSION 3.16)
project(GqaAttentionBaselineExperiment LANGUAGES CXX)
include(GNUInstallDirs)

set(SOC_VERSION "" CACHE STRING "Ascend SoC version")
if(NOT SOC_VERSION)
    message(FATAL_ERROR "SOC_VERSION not set. Use scripts/build.sh or pass -DSOC_VERSION=ascend910b4")
endif()
set(ASCEND_CANN_PACKAGE_PATH "/usr/local/Ascend/ascend-toolkit/latest" CACHE STRING "CANN root")
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()
if(EXISTS ${ASCEND_CANN_PACKAGE_PATH}/aarch64-linux/tikcpp/ascendc_kernel_cmake)
    set(ASCENDC_CMAKE_DIR ${ASCEND_CANN_PACKAGE_PATH}/aarch64-linux/tikcpp/ascendc_kernel_cmake)
elseif(EXISTS ${ASCEND_CANN_PACKAGE_PATH}/x86_64-linux/tikcpp/ascendc_kernel_cmake)
    set(ASCENDC_CMAKE_DIR ${ASCEND_CANN_PACKAGE_PATH}/x86_64-linux/tikcpp/ascendc_kernel_cmake)
elseif(EXISTS ${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake)
    set(ASCENDC_CMAKE_DIR ${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake)
else()
    message(FATAL_ERROR "ascendc_kernel_cmake not found under ${ASCEND_CANN_PACKAGE_PATH}")
endif()
include(${ASCENDC_CMAKE_DIR}/ascendc.cmake)

ascendc_library(ascendc_gqa_kernels SHARED op_kernel/gqa_attention_baseline_kernel.cpp)
ascendc_include_directories(ascendc_gqa_kernels PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/op_kernel)
set(_HCC_CXX_ROOT "${ASCEND_CANN_PACKAGE_PATH}/tools/hcc/aarch64-target-linux-gnu/include/c++/7.3.0")
if(EXISTS "${_HCC_CXX_ROOT}")
    ascendc_include_directories(ascendc_gqa_kernels PRIVATE "${_HCC_CXX_ROOT}" "${_HCC_CXX_ROOT}/aarch64-target-linux-gnu")
endif()

add_executable(gqa_attention_baseline_standalone verify_kernel_launch/gqa_attention_baseline_standalone.cpp)
target_compile_options(gqa_attention_baseline_standalone PRIVATE -O2 -std=c++17 -D_GLIBCXX_USE_CXX11_ABI=0 -Wall -Wextra)
target_include_directories(gqa_attention_baseline_standalone PRIVATE ${CMAKE_BINARY_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/op_kernel)
target_link_libraries(gqa_attention_baseline_standalone PRIVATE ascendcl ascendc_gqa_kernels dl)

execute_process(COMMAND ${CMAKE_COMMAND} -E env TORCH_DEVICE_BACKEND_AUTOLOAD=0 python3 -c "import torch; print(torch.__path__[0], end='')"
    OUTPUT_VARIABLE TORCH_PATH RESULT_VARIABLE TORCH_FIND_RET OUTPUT_STRIP_TRAILING_WHITESPACE)
if(TORCH_FIND_RET EQUAL 0 AND EXISTS "${TORCH_PATH}")
    set_source_files_properties(torch_extension/gqa_attention_torch_register.asc PROPERTIES LANGUAGE CXX)
    add_library(gqa_attention_torch_register SHARED torch_extension/gqa_attention_torch_register.asc)
    target_compile_options(gqa_attention_torch_register PRIVATE -O2 -std=c++17 -D_GLIBCXX_USE_CXX11_ABI=1 -x c++)
    target_include_directories(gqa_attention_torch_register PRIVATE ${CMAKE_BINARY_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/op_kernel ${ASCEND_CANN_PACKAGE_PATH}/include ${TORCH_PATH}/include)
    target_link_directories(gqa_attention_torch_register PRIVATE ${TORCH_PATH}/lib)
    target_link_libraries(gqa_attention_torch_register PRIVATE torch_cpu torch c10 ascendcl ascendc_gqa_kernels)
    install(TARGETS gqa_attention_torch_register DESTINATION lib)
else()
    message(WARNING "PyTorch not found. Torch extension will not be built.")
endif()
install(TARGETS ascendc_gqa_kernels DESTINATION lib)
install(TARGETS gqa_attention_baseline_standalone RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})

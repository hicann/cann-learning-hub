cmake_minimum_required(VERSION 3.16)
project(GEMM_acl LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(ACL_C_STUB "Build a host-only diagnostic executable when CANN is unavailable" OFF)

set(ASCEND_HOME "$ENV{ASCEND_HOME}" CACHE PATH "Ascend toolkit root")
if(NOT ASCEND_HOME)
  set(ASCEND_HOME "$ENV{ASCEND_HOME_PATH}")
endif()

find_path(ACL_INCLUDE_DIR acl/acl.h HINTS "${ASCEND_HOME}/include" "${ASCEND_HOME}/runtime/include")
find_path(ACLNN_INCLUDE_DIR aclnnop/aclnn_gemm.h HINTS "${ASCEND_HOME}/include" "${ASCEND_HOME}/runtime/include")
find_library(ASCENDCL_LIBRARY NAMES ascendcl HINTS "${ASCEND_HOME}/lib64" "${ASCEND_HOME}/runtime/lib64")
find_library(OPAPI_LIBRARY NAMES opapi HINTS "${ASCEND_HOME}/lib64" "${ASCEND_HOME}/runtime/lib64")
find_library(NNOPBASE_LIBRARY NAMES nnopbase HINTS "${ASCEND_HOME}/lib64" "${ASCEND_HOME}/runtime/lib64")

if(ACL_C_STUB)
  add_executable(gemm_acl src/stub_main.cpp)
  target_compile_definitions(gemm_acl PRIVATE ACL_C_STUB=1)
else()
  if(NOT ACL_INCLUDE_DIR OR NOT ACLNN_INCLUDE_DIR OR NOT ASCENDCL_LIBRARY OR NOT OPAPI_LIBRARY OR NOT NNOPBASE_LIBRARY)
    message(FATAL_ERROR "CANN not found. Source set_env.sh or configure -DASCEND_HOME=/path/to/ascend-toolkit. For local syntax-only build use -DACL_C_STUB=ON.")
  endif()
  add_executable(gemm_acl src/main.cpp)
  target_include_directories(gemm_acl PRIVATE include "${ACL_INCLUDE_DIR}" "${ACLNN_INCLUDE_DIR}")
  target_link_libraries(gemm_acl PRIVATE "${OPAPI_LIBRARY}" "${ASCENDCL_LIBRARY}" "${NNOPBASE_LIBRARY}" dl rt pthread)
  set_target_properties(gemm_acl PROPERTIES BUILD_RPATH "${ASCEND_HOME}/lib64")
endif()

cmake_minimum_required(VERSION 3.16)
project(SpMV_acl LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(ACL_C_STUB "Build a host-only diagnostic executable when CANN is unavailable" OFF)

set(ASCEND_HOME "$ENV{ASCEND_HOME}" CACHE PATH "Ascend toolkit root")
if(NOT ASCEND_HOME)
  set(ASCEND_HOME "$ENV{ASCEND_HOME_PATH}")
endif()
set(OPS_SPARSE_ROOT "$ENV{OPS_SPARSE_ROOT}" CACHE PATH "ops-sparse installation root")
set(OPS_SPARSE_SEARCH_ROOTS
  "${OPS_SPARSE_ROOT}"
  "${ASCEND_HOME}")
find_path(ACL_INCLUDE_DIR acl/acl.h HINTS "${ASCEND_HOME}/include" "${ASCEND_HOME}/runtime/include")
find_path(OPS_SPARSE_INCLUDE_DIR cann_ops_sparse.h HINTS ${OPS_SPARSE_SEARCH_ROOTS} PATH_SUFFIXES include)
find_library(ASCENDCL_LIBRARY NAMES ascendcl HINTS "${ASCEND_HOME}/lib64" "${ASCEND_HOME}/runtime/lib64")
find_library(OPS_SPARSE_LIBRARY NAMES ops_sparse HINTS ${OPS_SPARSE_SEARCH_ROOTS} PATH_SUFFIXES lib64)
find_library(C_SEC_LIBRARY NAMES c_sec HINTS "${ASCEND_HOME}/lib64" "${ASCEND_HOME}/runtime/lib64")

if(ACL_C_STUB)
  add_executable(spmv_acl src/stub_main.cpp)
  target_compile_definitions(spmv_acl PRIVATE ACL_C_STUB=1)
else()
  if(NOT ACL_INCLUDE_DIR OR NOT OPS_SPARSE_INCLUDE_DIR OR NOT ASCENDCL_LIBRARY OR NOT OPS_SPARSE_LIBRARY)
    message(FATAL_ERROR "ACL/ops-sparse not found. Set ASCEND_HOME to the Toolkit root and OPS_SPARSE_ROOT to the directory containing include/ and lib64/.")
  endif()
  add_executable(spmv_acl src/main.cpp)
  target_include_directories(spmv_acl PRIVATE include "${ACL_INCLUDE_DIR}" "${OPS_SPARSE_INCLUDE_DIR}")
  target_link_libraries(spmv_acl PRIVATE "${OPS_SPARSE_LIBRARY}" "${ASCENDCL_LIBRARY}")
  if(C_SEC_LIBRARY)
    target_link_libraries(spmv_acl PRIVATE "${C_SEC_LIBRARY}")
  endif()
  get_filename_component(OPS_SPARSE_LIBRARY_DIR "${OPS_SPARSE_LIBRARY}" DIRECTORY)
  set_target_properties(spmv_acl PROPERTIES BUILD_RPATH "${OPS_SPARSE_LIBRARY_DIR};${ASCEND_HOME}/lib64")
endif()

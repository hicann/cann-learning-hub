cmake_minimum_required(VERSION 3.16)
project(hccl_spmv LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

option(HCCL_SPMV_STUB "Build host-only fallback when CANN is unavailable" OFF)
set(ASCEND_HOME "$ENV{ASCEND_HOME}" CACHE PATH "Ascend toolkit root")
if(NOT ASCEND_HOME)
  set(ASCEND_HOME "$ENV{ASCEND_HOME_PATH}")
endif()

find_path(ACL_INCLUDE_DIR acl/acl.h HINTS "${ASCEND_HOME}/include" "${ASCEND_HOME}/runtime/include")
find_path(HCCL_INCLUDE_DIR hccl/hccl.h HINTS "${ASCEND_HOME}/include" "${ASCEND_HOME}/runtime/include")
find_library(ASCENDCL_LIBRARY NAMES ascendcl HINTS "${ASCEND_HOME}/lib64" "${ASCEND_HOME}/runtime/lib64")
find_library(HCCL_LIBRARY NAMES hccl HINTS "${ASCEND_HOME}/lib64" "${ASCEND_HOME}/runtime/lib64")
find_library(ACL_RTC_LIBRARY NAMES acl_rtc HINTS "${ASCEND_HOME}/lib64" "${ASCEND_HOME}/runtime/lib64")
if(NOT HCCL_SPMV_STUB AND ACL_INCLUDE_DIR AND HCCL_INCLUDE_DIR AND ASCENDCL_LIBRARY AND HCCL_LIBRARY AND ACL_RTC_LIBRARY)
  set(HCCL_SPMV_HAS_CANN 1)
else()
  set(HCCL_SPMV_HAS_CANN 0)
endif()
if(NOT HCCL_SPMV_STUB AND NOT HCCL_SPMV_HAS_CANN)
  message(FATAL_ERROR "HCCL_SPMV_STUB=OFF requires ACL, HCCL and acl_rtc. Source CANN set_env.sh; use -DHCCL_SPMV_STUB=ON only for Host diagnostics.")
endif()
message(STATUS "ASCEND_HOME=${ASCEND_HOME}")
message(STATUS "ACL include=${ACL_INCLUDE_DIR}")
message(STATUS "HCCL include=${HCCL_INCLUDE_DIR}")
message(STATUS "HCCL backend=${HCCL_SPMV_HAS_CANN} (1=ACL+HCCL, 0=host stub)")
set(HCCL_SPMV_HAS_CANN "${HCCL_SPMV_HAS_CANN}" CACHE INTERNAL "Resolved real HCCL backend")

add_library(hccl_spmv_core
  src/spmv_cpu.cpp src/hccl_context.cpp src/device_spmv.cpp src/spmv_distributed_hccl.cpp)
target_include_directories(hccl_spmv_core PUBLIC include)
target_compile_definitions(hccl_spmv_core PUBLIC HCCL_SPMV_HAS_CANN=${HCCL_SPMV_HAS_CANN})
if(HCCL_SPMV_HAS_CANN)
  target_include_directories(hccl_spmv_core PUBLIC "${ACL_INCLUDE_DIR}" "${HCCL_INCLUDE_DIR}")
  target_link_libraries(hccl_spmv_core PUBLIC "${ASCENDCL_LIBRARY}" "${HCCL_LIBRARY}" "${ACL_RTC_LIBRARY}")
  target_compile_definitions(hccl_spmv_core PRIVATE HCCL_SPMV_KERNEL_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}/kernels/spmv_local_fp32_kernel.cpp")
  set_target_properties(hccl_spmv_core PROPERTIES BUILD_RPATH "${ASCEND_HOME}/lib64;${ASCEND_HOME}/runtime/lib64")
endif()
add_executable(hccl_spmv src/main.cpp)
target_link_libraries(hccl_spmv PRIVATE hccl_spmv_core)
if(NOT HCCL_SPMV_HAS_CANN)
  message(STATUS "CANN ACL/HCCL not found: building host-only HCCL flow stub")
endif()

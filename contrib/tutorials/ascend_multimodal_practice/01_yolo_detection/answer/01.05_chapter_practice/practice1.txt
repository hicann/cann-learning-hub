# ===== 完整可运行版本 =====
import numpy as np
import cv2
import matplotlib.pyplot as plt

detections = [
    {"box": (50, 50, 200, 180), "label": "apple",  "conf": 0.95},
    {"box": (250, 80, 380, 220), "label": "banana", "conf": 0.88},
    {"box": (100, 250, 300, 380), "label": "orange", "conf": 0.76},
]

# 浅灰色背景
canvas = np.full((450, 450, 3), 230, dtype=np.uint8)

# 统一绿色（BGR）
color = (0, 200, 0)

for det in detections:
    x1, y1, x2, y2 = det["box"]
    # 画边界框：cv2.rectangle(img, 左上角(x1,y1), 右下角(x2,y2), color, thickness)
    cv2.rectangle(canvas, (x1, y1), (x2, y2), color, 3)
    # 写标签：在框左上角上方一点（y1-8 避免压在框线上）
    label_text = f"{det['label']} {det['conf']}"
    cv2.putText(canvas, label_text, (x1, y1 - 8),
                cv2.FONT_HERSHEY_SIMPLEX, 0.6, color, 2)

plt.imshow(cv2.cvtColor(canvas, cv2.COLOR_BGR2RGB))
plt.title("模拟 YOLO 多目标检测")
plt.axis('off')
plt.show()

# 关键点：
# - 遍历 detections 列表，对每个目标分别画框 + 写标签
# - box 格式 (x1,y1,x2,y2)：前两个是左上角，后两个是右下角，直接传给 cv2.rectangle
# - 标签 y 坐标用 y1-8 让文字在框线上方（不遮挡框内内容），若 y1 太小可改为 y1+20 写在框内
# - 这正是 YOLO 库 result.plot() 方法内部做的事情，第 2 章会直接调用封装好的方法
